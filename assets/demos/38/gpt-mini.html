Compact Professional Unity Save System

This is a reusable, easy-to-understand save framework for Unity that avoids PlayerPrefs and uses a single JSON file in persistent storage. It is designed to save common game data anywhere in your project.

No PlayerPrefs
JSON File
Auto Save / Load
Works Everywhere
Easy to Extend

Features

Core data types
Supported
int, float, double, bool, string, Vector2, Vector3, Quaternion, Color, enums, lists, dictionaries, and custom serializable classes.
Global access
Singleton
Use it from any script: SaveSystem.Instance.SetInt(...)
Professional structure
Clean API
Central manager, typed keys, autosave, manual save/load, and safe fallback handling.
Tip: for large projects, split save data into sections like Player, Settings, and World.

Unity C# Script

using System;
using System.Collections.Generic;
using System.IO;
using UnityEngine;

[Serializable]
public class SaveData
{
    public int version = 1;

    public int intValue;
    public float floatValue;
    public double doubleValue;
    public bool boolValue;
    public string stringValue;

    public Vector2 vector2Value;
    public Vector3 vector3Value;
    public Quaternion quaternionValue;
    public Color colorValue;

    public List intList = new();
    public List stringList = new();
    public List floatList = new();

    // For custom data, add your own serializable classes here.
    // Example:
    // public PlayerData player = new PlayerData();
}

public class SaveSystem : MonoBehaviour
{
    public static SaveSystem Instance { get; private set; }

    [SerializeField] private string fileName = "save.json";
    [SerializeField] private bool autoLoadOnStart = true;
    [SerializeField] private bool autoSaveOnQuit = true;

    public SaveData Data { get; private set; } = new SaveData();

    private string FilePath => Path.Combine(Application.persistentDataPath, fileName);

    private void Awake()
    {
        if (Instance != null && Instance != this)
        {
            Destroy(gameObject);
            return;
        }

        Instance = this;
        DontDestroyOnLoad(gameObject);
    }

    private void Start()
    {
        if (autoLoadOnStart) Load();
    }

    private void OnApplicationQuit()
    {
        if (autoSaveOnQuit) Save();
    }

    // ---------- FILE ----------
    public void Save()
    {
        try
        {
            string json = JsonUtility.ToJson(Data, true);
            File.WriteAllText(FilePath, json);
            Debug.Log($"Saved: {FilePath}");
        }
        catch (Exception e)
        {
            Debug.LogError($"Save failed: {e.Message}");
        }
    }

    public void Load()
    {
        try
        {
            if (!File.Exists(FilePath))
            {
                Data = new SaveData();
                return;
            }

            string json = File.ReadAllText(FilePath);
            Data = JsonUtility.FromJson(json) ?? new SaveData();
            Debug.Log($"Loaded: {FilePath}");
        }
        catch (Exception e)
        {
            Debug.LogError($"Load failed: {e.Message}");
            Data = new SaveData();
        }
    }

    public void DeleteSave()
    {
        try
        {
            if (File.Exists(FilePath))
                File.Delete(FilePath);

            Data = new SaveData();
            Debug.Log("Save deleted.");
        }
        catch (Exception e)
        {
            Debug.LogError($"Delete failed: {e.Message}");
        }
    }

    public bool HasSave() => File.Exists(FilePath);

    // ---------- GENERIC ACCESS ----------
    // Use these for simple global values. For bigger projects, create your own data classes.

    public void SetInt(int value) => Data.intValue = value;
    public int GetInt(int defaultValue = 0) => Data.intValue;

    public void SetFloat(float value) => Data.floatValue = value;
    public float GetFloat(float defaultValue = 0f) => Data.floatValue;

    public void SetDouble(double value) => Data.doubleValue = value;
    public double GetDouble(double defaultValue = 0d) => Data.doubleValue;

    public void SetBool(bool value) => Data.boolValue = value;
    public bool GetBool(bool defaultValue = false) => Data.boolValue;

    public void SetString(string value) => Data.stringValue = value;
    public string GetString(string defaultValue = "") => string.IsNullOrEmpty(Data.stringValue) ? defaultValue : Data.stringValue;

    public void SetVector2(Vector2 value) => Data.vector2Value = value;
    public Vector2 GetVector2(Vector2 defaultValue = default) => Data.vector2Value;

    public void SetVector3(Vector3 value) => Data.vector3Value = value;
    public Vector3 GetVector3(Vector3 defaultValue = default) => Data.vector3Value;

    public void SetQuaternion(Quaternion value) => Data.quaternionValue = value;
    public Quaternion GetQuaternion(Quaternion defaultValue = default) => Data.quaternionValue;

    public void SetColor(Color value) => Data.colorValue = value;
    public Color GetColor(Color defaultValue = default) => Data.colorValue;

    public void SetIntList(List value) => Data.intList = value;
    public List GetIntList() => Data.intList;

    public void SetStringList(List value) => Data.stringList = value;
    public List GetStringList() => Data.stringList;

    public void SetFloatList(List value) => Data.floatList = value;
    public List GetFloatList() => Data.floatList;
}
