⚙️ SaveManager - Main Controller
Core

The central manager that handles all save operations. Single instance, thread-safe, with automatic backup and versioning.

C# — SaveManager.cs 
using System;
using System.IO;
using System.Collections.Generic;
using UnityEngine;
using UnityEngine.Networking;
using Newtonsoft.Json;

public enum SaveFormat
{
    JSON,
    Binary
}

public enum SaveSlot
{
    Slot1,
    Slot2,
    Slot3,
    Custom
}

[Serializable]
public class SaveData
{
    public string Version;
    public DateTime SavedAt;
    public DateTime LastLoaded;
    public string SlotName;
    public int PlayTime;
    public int Level;
    public Dictionary<string, object> Data;

    public SaveData()
    {
        Version = "1.0.0";
        SavedAt = DateTime.UtcNow;
        LastLoaded = DateTime.UtcNow;
        SlotName = "Default";
        PlayTime = 0;
        Level = 1;
        Data = new Dictionary<string, object>();
    }

    public void UpdateMetadata(string slotName, int playTime, int level)
    {
        SlotName = slotName;
        PlayTime = playTime;
        Level = level;
        SavedAt = DateTime.UtcNow;
    }
}

public class SaveManager : MonoBehaviour
{
    public static SaveManager Instance { get; private set; }

    private const string SAVE_DIR = "Saves";
    private const string BACKUP_DIR = "Saves/Backups";
    private const string VERSION = "1.0.0";

    private SaveFormat _format = SaveFormat.JSON;
    private bool _encrypt = false;
    private string _encryptionKey = "your-32-character-encryption-key-here!!";

    private Dictionary<string, SaveData> _cache = new();

    void Awake()
    {
        if (Instance != null && Instance != this)
        {
            Destroy(gameObject);
            return;
        }
        Instance = this;
        DontDestroyOnLoad(gameObject);
        Directory.CreateDirectory(Path.Combine(Application.persistentDataPath, SAVE_DIR));
        Directory.CreateDirectory(Path.Combine(Application.persistentDataPath, BACKUP_DIR));
    }

    public string GetSavePath(SaveSlot slot = SaveSlot.Slot1, string customName = null)
    {
        string fileName = customName ?? slot.ToString().ToLower();
        string extension = _format == SaveFormat.JSON ? ".json" : ".bin";
        return Path.Combine(Application.persistentDataPath, SAVE_DIR, fileName + extension);
    }

    public Task<bool> SaveAsync(SaveData data, SaveSlot slot = SaveSlot.Slot1, string customName = null)
    {
        return Task.Run(() =>
        {
            try
            {
                string path = GetSavePath(slot, customName);
                string content;

                if (_format == SaveFormat.JSON)
                {
                    content = JsonConvert.SerializeObject(data, Formatting.Indented);
                }
                else
                {
                    content = Convert.ToBase64String(SerializeBinary(data));
                }

                if (_encrypt)
                {
                    content = Encrypt(content);
                }

                File.WriteAllText(path, content);
                _cache[customName ?? slot.ToString()] = data;
                CreateBackup(path);
                return true;
            }
            catch (Exception e)
            {
                Debug.LogError($"[SaveManager] Save failed: {e.Message}");
                return false;
            }
        });
    }

    public Task<SaveData> LoadAsync(SaveSlot slot = SaveSlot.Slot1, string customName = null)
    {
        return Task.Run(() =>
        {
            try
            {
                string path = GetSavePath(slot, customName);
                if (!File.Exists(path)) return null;

                string content = File.ReadAllText(path);

                if (_encrypt)
                {
                    content = Decrypt(content);
                }

                SaveData data;
                if (_format == SaveFormat.JSON)
                {
                    data = JsonConvert.DeserializeObject<SaveData>(content);
                }
                else
                {
                    data = DeserializeBinary(Convert.FromBase64String(content));
                }

                if (data != null)
                {
                    data.LastLoaded = DateTime.UtcNow;
                    _cache[customName ?? slot.ToString()] = data;
                }
                return data;
            }
            catch (Exception e)
            {
                Debug.LogError($"[SaveManager] Load failed: {e.Message}");
                return null;
            }
        });
    }

    public Task<bool> DeleteAsync(SaveSlot slot = SaveSlot.Slot1, string customName = null)
    {
        return Task.Run(() =>
        {
            string path = GetSavePath(slot, customName);
            if (File.Exists(path))
            {
                File.Delete(path);
                string key = customName ?? slot.ToString();
                if (_cache.ContainsKey(key)) _cache.Remove(key);
                return true;
            }
            return false;
        });
    }

    public Task<List<string>> GetAllSavesAsync()
    {
        return Task.Run(() =>
        {
            string savePath = Path.Combine(Application.persistentDataPath, SAVE_DIR);
            if (!Directory.Exists(savePath)) return new List<string>();
            return new List<string>(Directory.GetFiles(savePath));
        });
    }

    private void CreateBackup(string filePath)
    {
        try
        {
            string fileName = Path.GetFileName(filePath);
            string backupPath = Path.Combine(Application.persistentDataPath, BACKUP_DIR,
                $"{fileName}_backup_{DateTime.UtcNow:yyyyMMdd_HHmmss}");
            File.Copy(filePath, backupPath);
        }
        catch { }
    }

    private string Encrypt(string text)
    {
        using (var aes = System.Security.Cryptography.AES.Create())
        {
            aes.Key = System.Text.Encoding.UTF8.GetBytes(_encryptionKey.PadRight(32).Substring(0, 32));
            aes.IV = System.Text.Encoding.UTF8.GetBytes(_encryptionKey.PadRight(16).Substring(0, 16));
            var encryptor = aes.CreateEncryptor();
            var encrypted = encryptor.TransformFinalBlock(
                System.Text.Encoding.UTF8.GetBytes(text), 0, text.Length);
            return Convert.ToBase64String(encrypted);
        }
    }

    private string Decrypt(string ciphertext)
    {
        using (var aes = System.Security.Cryptography.AES.Create())
        {
            aes.Key = System.Text.Encoding.UTF8.GetBytes(_encryptionKey.PadRight(32).Substring(0, 32));
            aes.IV = System.Text.Encoding.UTF8.GetBytes(_encryptionKey.PadRight(16).Substring(0, 16));
            var decryptor = aes.CreateDecryptor();
            var encryptedBytes = Convert.FromBase64String(ciphertext);
            var decrypted = decryptor.TransformFinalBlock(encryptedBytes, 0, encryptedBytes.Length);
            return System.Text.Encoding.UTF8.GetString(decrypted);
        }
    }

    private byte[] SerializeBinary(object obj)
    {
        using (var ms = new MemoryStream())
        {
            new BinaryFormatter().Serialize(ms, obj);
            return ms.ToArray();
        }
    }

    private object DeserializeBinary(byte[] data)
    {
        using (var ms = new MemoryStream(data))
        {
            return new BinaryFormatter().Deserialize(ms);
        }
    }
}
📦 SaveSystem - Easy API Wrapper
Easy Use

A simple, compact wrapper for quick save/load operations. Perfect for small projects or rapid prototyping.

C# — SaveSystem.cs 
using System;
using System.Collections.Generic;
using UnityEngine;
using Newtonsoft.Json;

public static class SaveSystem
{
    private static string SavePath =>
        Path.Combine(Application.persistentDataPath, "save.json");

    public static void Save<T>(T data, string filename = "save")
    {
        string json = JsonConvert.SerializeObject(data, Formatting.Indented);
        File.WriteAllText(Path.Combine(Application.persistentDataPath, $"{filename}.json"), json);
        Debug.Log($"[SaveSystem] Saved: {filename}");
    }

    public static T Load<T>(string filename = "save")
    {
        string path = Path.Combine(Application.persistentDataPath, $"{filename}.json");
        if (!File.Exists(path)) return default;
        string json = File.ReadAllText(path);
        return JsonConvert.DeserializeObject<T>(json);
    }

    public static bool Exists(string filename = "save")
    {
        return File.Exists(Path.Combine(Application.persistentDataPath, $"{filename}.json"));
    }

    public static void Delete(string filename = "save")
    {
        string path = Path.Combine(Application.persistentDataPath, $"{filename}.json");
        if (File.Exists(path)) File.Delete(path);
    }

    public static List<string> GetAllFilenames()
    {
        var dir = Application.persistentDataPath;
        var files = Directory.GetFiles(dir, "*.json");
        var names = new List<string>();
        foreach (var f in files)
        {
            names.Add(Path.GetFileNameWithoutExtension(f));
        }
        return names;
    }
}

// ===== Vector2 Converter =====
public class Vector2Converter : JsonConverter<Vector2>
{
    public override void WriteJson(JsonWriter writer, Vector2 value, JsonSerializer serializer)
    {
        writer.WriteStartObject();
        writer.WritePropertyName("x");
        writer.WriteValue(value.x);
        writer.WritePropertyName("y");
        writer.WriteValue(value.y);
        writer.WriteEndObject();
    }

    public override Vector2 ReadJson(JsonReader reader, Type objectType, Vector2 existingValue,
        bool hasExistingValue, JsonSerializer serializer)
    {
        var obj = JObject.Load(reader);
        return new Vector2((float)obj["x"], (float)obj["y"]);
    }
}

    public string playerName;

        playerName = "Player";

        cameraOffset = Vector2.zero;
        playerPosition = Vector3.zero;
        playerRotation = Vector3.zero;
        playerQuaternion = Quaternion.identity;
        customVector = Vector4.zero;
        mainColor = Color.white;
        uiColor = Color32.white;
        windowRect = new Rect(0, 0, 800, 600);

        playerClass = PlayerClass.Warrior;
        difficulty = Difficulty.Normal;

        stats = new PlayerStats();
        inventory = new List<InventoryItem>();
        unlockedLevels = new string[] { "level1" };
        achievements = new int[] { 0 };
        statsDictionary = new Dictionary<string, int>();
        checkpointPositions = new Dictionary<string, Vector3>();

        firstPlayed = DateTime.UtcNow;
        lastPlayed = DateTime.UtcNow;
        totalPlayTime = 0;
    }

    // ===== Helper Methods =====
    public void AddToInventory(string itemId, string itemName, int quantity = 1)
    {
        var existing = inventory.Find(i => i.itemId == itemId);
        if (existing != null)
        {
            existing.quantity += quantity;
        }
        else
        {
            inventory.Add(new InventoryItem(itemId, itemName, quantity));
        }
    }

    public void AddExperience(float amount)
    {
        stats.experience += amount;
        while (stats.experience >= stats.level * 100)
        {
            stats.experience -= stats.level * 100;
            stats.level++;
            stats.maxHealth += 10;
            stats.health = stats.maxHealth;
            stats.maxMana += 5;
            stats.mana = stats.maxMana;
        }
    }
}

📦 Cloud Save Integration ▼

Extend the system to support cloud saves using Unity Cloud Save or any REST API.

C# — Cloud Save 
using UnityEngine.Networking;
using System.Threading.Tasks;

public class CloudSaveManager
{
    private const string API_URL = "https://api.yourgame.com/saves";

    public static async Task<bool> UploadSaveAsync(string saveData, string userId)
    {
        using (var request = new UnityWebRequest(API_URL, "POST"))
        {
            var json = System.Text.Json.JsonSerializer.Serialize(new { userId, data = saveData });
            var body = System.Text.Encoding.UTF8.GetBytes(json);
            request.uploadHandler = new UploadHandlerRaw(body);
            request.downloadHandler = new DownloadHandlerBuffer();
            request.SetRequestHeader("Content-Type", "application/json");

            await request.SendWebRequest();
            return request.result == UnityWebRequest.Result.Success;
        }
    }

    public static async Task<string> DownloadSaveAsync(string userId)
    {
        using (var request = new UnityWebRequest($"{API_URL}/{userId}"))
        {
            request.method = "GET";
            await request.SendWebRequest();

            if (request.result == UnityWebRequest.Result.Success)
            {
                return request.downloadHandler.text;
            }
            return null;
        }
    }
}
🔄 Save Versioning & Migration ▼

Handle save file format changes with automatic migration.

C# — Migration 
public class SaveMigration
{
    public static SaveData Migrate(SaveData data)
    {
        switch (data.Version)
        {
            case "1.0.0":
                // Migrate from v1 to v2
                data.Data["newFeature"] = false;
                data.Version = "2.0.0";
                goto case "2.0.0";

            case "2.0.0":
                // Migrate from v2 to v3
                if (data.Data.ContainsKey("oldStat"))
                {
                    data.Data["newStat"] = data.Data["oldStat"];
                    data.Data.Remove("oldStat");
                }
                data.Version = "3.0.0";
                break;

            default:
                // Already latest version
                break;
        }
        return data;
    }
}

using System.Runtime.Serialization.Formatters.Binary;

[Serializable]
public class BinarySaveManager
{
    public static void Save<T>(T data, string path)
    {
        using (var ms = new MemoryStream())
        {
            var formatter = new BinaryFormatter();
            formatter.Serialize(ms, data);
            File.WriteAllBytes(path, ms.ToArray());
        }
    }

    public static T Load<T>(string path)
    {
        if (!File.Exists(path)) return default;
        using (var ms = new MemoryStream(File.ReadAllBytes(path)))
        {
            var formatter = new BinaryFormatter();
            return (T)formatter.Deserialize(ms);
        }
    }
}
🎮 Unity Component Save/Load ▼

Save and restore entire Unity component states.

C# — Component Save 
using UnityEngine;
using Newtonsoft.Json;

public static class ComponentSaver
{
    public static Dictionary<string, object> SaveComponent<T>(T component) where T : Component
    {
        var data = new Dictionary<string, object>();
        var type = component.GetType();

        foreach (var prop in type.GetProperties())
        {
            if (prop.CanRead && prop.CanWrite)
            {
                var value = prop.GetValue(component);
                data[prop.Name] = value;
            }
        }

        return data;
    }

    public static void LoadComponent<T>(T component, Dictionary<string, object> data) where T : Component
    {
        var type = component.GetType();

        foreach (var kvp in data)
        {
            var prop = type.GetProperty(kvp.Key);
            if (prop != null && prop.CanWrite)
            {
                try
                {
                    prop.SetValue(component, kvp.Value);
                }
                catch
                {
                    Debug.LogWarning($"[ComponentSaver] Failed to set {kvp.Key}");
                }
            }
        }
    }
}
