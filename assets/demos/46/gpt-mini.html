FIFA Card Duel

2D Sprite-Based Portrait Only Async Multiplayer Unity / C#

Game Summary

A turn-based football card battler for mobile where players build a 40-card deck, place cards across 5 zones (GK-DEF-MID-ATT-ATT), and resolve attacks with stat-based combat. The design supports asynchronous multiplayer, quick matches, ranked divisions, and draft mode. Monetization is cosmetic-only; energy gates session frequency without pay-to-win benefits.

Turn loop: Draw → Place → Attack → End. Each async turn has a 90-second action window.

Included Deliverables

  • Card.cs
  • GameManager.cs
  • CombatSystem.cs
  • UIManager.cs
  • 50-card JSON database
This response provides a self-contained Unity-oriented implementation blueprint and data set in C# and JSON form.

using System;
using UnityEngine;

public enum CardType
{
    Player,
    Tactic,
    Reaction
}

public enum CardRarity
{
    Bronze = 1,
    Silver = 2,
    Gold = 3,
    Special = 4
}

public enum ZoneType
{
    GK,
    DEF,
    MID,
    ATT1,
    ATT2
}

[Serializable]
public class Card
{
    public string id;
    public string name;
    public CardType type;
    public CardRarity rarity;
    [Range(1, 5)] public int stars = 1;

    // Player stats
    [Range(0, 99)] public int PAC;
    [Range(0, 99)] public int SHO;
    [Range(0, 99)] public int PAS;
    [Range(0, 99)] public int DRI;
    [Range(0, 99)] public int DEF;
    [Range(0, 99)] public int PHY;

    // Meta
    public string club;
    public string nation;
    public string league;
    public string formation; // for Tactic cards, e.g. "4-3-3"
    public string reactionEffect; // for Reaction cards, e.g. "Intercept", "Block", "Counter"
    public string description;
    public Sprite artwork;

    public int ManaCost
    {
        get
        {
            int baseCost = stars + ((int)rarity - 1);
            if (type == CardType.Tactic) baseCost += 1;
            if (type == CardType.Reaction) baseCost += 1;
            return Mathf.Clamp(baseCost, 1, 8);
        }
    }

    public int Overall
    {
        get
        {
            int total = PAC + SHO + PAS + DRI + DEF + PHY;
            return Mathf.RoundToInt(total / 6f);
        }
    }

    public bool IsPlayer => type == CardType.Player;
    public bool IsTactic => type == CardType.Tactic;
    public bool IsReaction => type == CardType.Reaction;

    public string DisplayStatLine()
    {
        return $"PAC {PAC} | SHO {SHO} | PAS {PAS} | DRI {DRI} | DEF {DEF} | PHY {PHY}";
    }

    public Color RarityColor()
    {
        switch (rarity)
        {
            case CardRarity.Bronze: return new Color(0.79f, 0.55f, 0.35f);
            case CardRarity.Silver: return new Color(0.82f, 0.84f, 0.88f);
            case CardRarity.Gold: return new Color(0.95f, 0.76f, 0.31f);
            case CardRarity.Special: return new Color(0.78f, 0.42f, 1f);
            default: return Color.white;
        }
    }

    public static Card Clone(Card c)
    {
        return new Card
        {
            id = c.id,
            name = c.name,
            type = c.type,
            rarity = c.rarity,
            stars = c.stars,
            PAC = c.PAC,
            SHO = c.SHO,
            PAS = c.PAS,
            DRI = c.DRI,
            DEF = c.DEF,
            PHY = c.PHY,
            club = c.club,
            nation = c.nation,
            league = c.league,
            formation = c.formation,
            reactionEffect = c.reactionEffect,
            description = c.description,
            artwork = c.artwork
        };
    }
}

using System;
using System.Collections.Generic;
using UnityEngine;

public enum MatchMode
{
    QuickMatch,
    Ranked,
    Draft
}

public enum TurnPhase
{
    Draw,
    Place,
    Attack,
    End
}

[Serializable]
public class BoardSlot
{
    public ZoneType zone;
    public Card placedCard;
    public bool occupied => placedCard != null;
}

[Serializable]
public class PlayerState
{
    public string playerId;
    public string displayName;
    public int energy = 10;
    public int divisionPoints;
    public int wins;
    public int losses;

    public List<Card> deck = new List<Card>();
    public List<Card> hand = new List<Card>();
    public List<Card> bench = new List<Card>();
    public List<BoardSlot> field = new List<BoardSlot>();

    public int goalsScored;
    public int goalsConceded;
    public int turnTimerSeconds = 90;
    public TurnPhase phase = TurnPhase.Draw;

    public PlayerState(string id, string name)
    {
        playerId = id;
        displayName = name;
        field.Add(new BoardSlot { zone = ZoneType.GK });
        field.Add(new BoardSlot { zone = ZoneType.DEF });
        field.Add(new BoardSlot { zone = ZoneType.MID });
        field.Add(new BoardSlot { zone = ZoneType.ATT1 });
        field.Add(new BoardSlot { zone = ZoneType.ATT2 });
    }
}

public class GameManager : MonoBehaviour
{
    public static GameManager Instance;

    [Header("Match")]
    public MatchMode mode;
    public PlayerState localPlayer;
    public PlayerState remotePlayer;
    public bool isAsyncMultiplayer = true;

    [Header("Timers")]
    public float turnDuration = 90f;
    public float quickMatchDuration = 180f;

    [Header("Deck Rules")]
    public int deckSize = 40;
    public int minPlayers = 11;
    public int benchSize = 5;
    public int maxHandSize = 7;

    [Header("AI")]
    public string aiArchetype = "Balanced"; // Defensive / Balanced / Aggressive

    public event Action<PlayerState> OnStateChanged;
    public event Action<string> OnLog;
    public event Action<PlayerState, Card> OnCardPlaced;
    public event Action<Card, Card, CombatResult> OnCombatResolved;
    public event Action<PlayerState> OnTurnAdvanced;
    public event Action<PlayerState> OnGoalScored;

    private System.Random rng = new System.Random();
    private CombatSystem combatSystem;

    void Awake()
    {
        if (Instance != null && Instance != this) { Destroy(gameObject); return; }
        Instance = this;
        combatSystem = GetComponent<CombatSystem>();
        if (combatSystem == null) combatSystem = gameObject.AddComponent<CombatSystem>();
    }

    public void StartMatch(PlayerState p1, PlayerState p2, MatchMode matchMode)
    {
        localPlayer = p1;
        remotePlayer = p2;
        mode = matchMode;

        SetupDeckAndHand(localPlayer);
        SetupDeckAndHand(remotePlayer);

        localPlayer.phase = TurnPhase.Draw;
        remotePlayer.phase = TurnPhase.Draw;

        Log($"Match started: {p1.displayName} vs {p2.displayName} ({matchMode})");
        OnStateChanged?.Invoke(localPlayer);
        OnStateChanged?.Invoke(remotePlayer);
    }

    void SetupDeckAndHand(PlayerState p)
    {
        Shuffle(p.deck);
        p.hand.Clear();
        p.bench.Clear();

        int initialHand = Mathf.Min(5, p.deck.Count);
        for (int i = 0; i < initialHand; i++)
            DrawCard(p);

        int benchCount = Mathf.Min(benchSize, p.deck.Count);
        for (int i = 0; i < benchCount; i++)
            p.bench.Add(p.deck[i]);
    }

    public void DrawCard(PlayerState p)
    {
        if (p.deck.Count == 0 || p.hand.Count >= maxHandSize) return;
        Card c = p.deck[0];
        p.deck.RemoveAt(0);
        p.hand.Add(c);
        Log($"{p.displayName} draws {c.name}");
        OnStateChanged?.Invoke(p);
    }

    public bool CanPlaceCard(Card c, ZoneType zone)
    {
        if (c == null) return false;
        if (c.type == CardType.Reaction) return false;
        if (c.type == CardType.Tactic) return zone == ZoneType.MID;
        if (c.type == CardType.Player)
        {
            if (zone == ZoneType.GK) return c.DEF >= 55 || c.Overall >= 60;
            if (zone == ZoneType.DEF) return c.DEF >= 45;
            if (zone == ZoneType.MID) return c.PAS + c.DRI >= 85;
            if (zone == ZoneType.ATT1 || zone == ZoneType.ATT2) return c.SHO + c.DRI >= 80;
        }
        return false;
    }

    public bool PlaceCard(PlayerState p, Card c, int zoneIndex)
    {
        if (zoneIndex < 0 || zoneIndex >= p.field.Count) return false;
        BoardSlot slot = p.field[zoneIndex];
        if (slot.occupied) return false;
        if (!p.hand.Contains(c)) return false;
        if (!CanPlaceCard(c, slot.zone)) return false;

        slot.placedCard = c;
        p.hand.Remove(c);
        OnCardPlaced?.Invoke(p, c);
        ApplyChemistry(p);
        Log($"{p.displayName} places {c.name} in {slot.zone}");
        OnStateChanged?.Invoke(p);
        return true;
    }

    void ApplyChemistry(PlayerState p)
    {
        for (int i = 0; i < p.field.Count; i++)
        {
            Card current = p.field[i].placedCard;
            if (current == null) continue;

            int boost = 0;
            if (i > 0 && p.field[i - 1].placedCard != null)
            {
                Card left = p.field[i - 1].placedCard;
                if (!string.IsNullOrEmpty(current.club) && current.club == left.club) boost += 1;
                if (!string.IsNullOrEmpty(current.nation) && current.nation == left.nation) boost += 1;
            }
            if (i < p.field.Count - 1 && p.field[i + 1].placedCard != null)
            {
                Card right = p.field[i + 1].placedCard;
                if (!string.IsNullOrEmpty(current.club) && current.club == right.club) boost += 1;
                if (!string.IsNullOrEmpty(current.nation) && current.nation == right.nation) boost += 1;
            }
            if (boost > 0)
            {
                current.PAC = Mathf.Min(99, current.PAC + boost);
                current.SHO = Mathf.Min(99, current.SHO + boost);
                current.PAS = Mathf.Min(99, current.PAS + boost);
                current.DRI = Mathf.Min(99, current.DRI + boost);
                current.DEF = Mathf.Min(99, current.DEF + boost);
                current.PHY = Mathf.Min(99, current.PHY + boost);
            }
        }
    }

    public CombatResult Attack(PlayerState attackerOwner, int attackerZoneIndex, PlayerState defenderOwner, int defenderZoneIndex)
    {
        if (attackerZoneIndex < 0 || attackerZoneIndex >= attackerOwner.field.Count) return CombatResult.Fail("Invalid attacker");
        if (defenderZoneIndex < 0 || defenderZoneIndex >= defenderOwner.field.Count) return CombatResult.Fail("Invalid defender");

        Card attacker = attackerOwner.field[attackerZoneIndex].placedCard;
        Card defender = defenderOwner.field[defenderZoneIndex].placedCard;
        if (attacker == null) return CombatResult.Fail("No attacker card");

        CombatResult result = combatSystem.ResolveAttack(attackerOwner, attacker, defenderOwner, defender, attackerZoneIndex, defenderZoneIndex);

        OnCombatResolved?.Invoke(attacker, defender, result);

        if (result.goal)
        {
            attackerOwner.goalsScored++;
            defenderOwner.goalsConceded++;
            OnGoalScored?.Invoke(attackerOwner);
            Log($"GOAL by {attackerOwner.displayName}!");
        }
        else
        {
            Log(result.message);
        }

        OnStateChanged?.Invoke(attackerOwner);
        OnStateChanged?.Invoke(defenderOwner);
        return result;
    }

    public void EndTurn(PlayerState p)
    {
        p.phase = TurnPhase.End;
        OnTurnAdvanced?.Invoke(p);
        Log($"{p.displayName} ends turn");
        OnStateChanged?.Invoke(p);
    }

    public void AdvanceToNextPhase(PlayerState p)
    {
        p.phase = (TurnPhase)(((int)p.phase + 1) % 4);
        if (p.phase == TurnPhase.Draw) DrawCard(p);
        OnTurnAdvanced?.Invoke(p);
        OnStateChanged?.Invoke(p);
    }

    public void TickAsyncTimer(PlayerState p, float deltaTime)
    {
        p.turnTimerSeconds = Mathf.Max(0, p.turnTimerSeconds - Mathf.RoundToInt(deltaTime));
        if (p.turnTimerSeconds == 0) EndTurn(p);
    }

    public bool IsDeckLegal(PlayerState p)
    {
        if (p.deck == null || p.deck.Count != deckSize) return false;
        int players = 0;
        foreach (var c in p.deck) if (c != null && c.type == CardType.Player) players++;
        return players >= minPlayers;
    }

    public List<Card> BuildDraftPool(List<Card> database, int count)
    {
        List<Card> pool = new List<Card>();
        for (int i = 0; i < count && database.Count > 0; i++)
        {
            int idx = rng.Next(database.Count);
            pool.Add(Card.Clone(database[idx]));
            database.RemoveAt(idx);
        }
        return pool;
    }

    void Shuffle(List<Card> list)
    {
        for (int i = 0; i < list.Count; i++)
        {
            int j = rng.Next(i, list.Count);
            (list[i], list[j]) = (list[j], list[i]);
        }
    }

    void Log(string msg)
    {
        Debug.Log(msg);
        OnLog?.Invoke(msg);
    }

    public bool CanPlayEnergy(PlayerState p, int cost) => p.energy >= cost;
    public void SpendEnergy(PlayerState p, int cost) => p.energy = Mathf.Max(0, p.energy - cost);
    public void RegenEnergy(PlayerState p)
    {
        p.energy = Mathf.Min(10, p.energy + 1);
    }
}

CombatSystem.cs

using System;
using UnityEngine;

[Serializable]
public class CombatResult
{
    public bool success;
    public bool goal;
    public int attackerScore;
    public int defenderScore;
    public int damage;
    public string message;

    public static CombatResult Fail(string msg)
    {
        return new CombatResult { success = false, goal = false, message = msg };
    }
}

public class CombatSystem : MonoBehaviour
{
    public CombatResult ResolveAttack(PlayerState attackerOwner, Card attacker, PlayerState defenderOwner, Card defender, int attackerZoneIndex, int defenderZoneIndex)
    {
        int attackStat = GetAttackStat(attacker, attackerZoneIndex);
        int defendStat = defender != null ? GetDefendStat(defender, defenderZoneIndex) : GetEmptyLaneDefense(defenderZoneIndex);

        int tacticBonusA = GetTacticBonus(attackerOwner, attackerZoneIndex);
        int tacticBonusD = GetTacticBonus(defenderOwner, defenderZoneIndex);

        int reactionBonusD = GetReactionBonus(defenderOwner, defenderZoneIndex);

        int attackerScore = attackStat + tacticBonusA + UnityEngine.Random.Range(0, 6);
        int defenderScore = defendStat + tacticBonusD + reactionBonusD + UnityEngine.Random.Range(0, 6);

        bool beat = attackerScore > defenderScore;
        bool isGoal = beat && defenderZoneIndex == 0;

        string msg;
        if (isGoal) msg = $"{attacker.name} beats GK and scores!";
        else if (beat) msg = $"{attacker.name} wins the duel.";
        else msg = defender != null ? $"{defender.name} stops {attacker.name}." : "Attack contained.";

        return new CombatResult
        {
            success = true,
            goal = isGoal,
            attackerScore = attackerScore,
            defenderScore = defenderScore,
            damage = beat ? 1 : 0,
            message = msg
        };
    }

    int GetAttackStat(Card c, int zoneIndex)
    {
        if (c == null) return 0;
        if (zoneIndex == 0) return c.PAS + c.PHY;
        if (zoneIndex == 1) return c.DEF + c.PHY;
        if (zoneIndex == 2) return c.PAS + c.DRI;
        return c.SHO + c.DRI + c.PAC / 2;
    }

    int GetDefendStat(Card c, int zoneIndex)
    {
        if (c == null) return 0;
        if (zoneIndex == 0) return c.DEF + c.PHY + 10;
        if (zoneIndex == 1) return c.DEF + c.PHY;
        if (zoneIndex == 2) return c.PAS + c.DEF;
        return c.DEF + c.PHY / 2;
    }

    int GetEmptyLaneDefense(int zoneIndex)
    {
        switch (zoneIndex)
        {
            case 0: return 35;
            case 1: return 25;
            case 2: return 20;
            default: return 15;
        }
    }

    int GetTacticBonus(PlayerState p, int zoneIndex)
    {
        int bonus = 0;
        foreach (var slot in p.field)
        {
            if (slot.placedCard != null && slot.placedCard.type == CardType.Tactic)
            {
                if (!string.IsNullOrEmpty(slot.placedCard.formation))
                {
                    if (slot.placedCard.formation == "4-3-3" && zoneIndex >= 2) bonus += 2;
                    if (slot.placedCard.formation == "5-4-1" && zoneIndex <= 1) bonus += 2;
                    if (slot.placedCard.formation == "4-2-3-1" && zoneIndex == 2) bonus += 2;
                }
            }
        }
        return bonus;
    }

    int GetReactionBonus(PlayerState p, int zoneIndex)
    {
        int bonus = 0;
        foreach (var slot in p.field)
        {
            if (slot.placedCard != null && slot.placedCard.type == CardType.Reaction)
            {
                if (slot.placedCard.reactionEffect == "Intercept" && zoneIndex == 2) bonus += 3;
                if (slot.placedCard.reactionEffect == "Block" && zoneIndex <= 1) bonus += 3;
                if (slot.placedCard.reactionEffect == "Counter" && zoneIndex == 0) bonus += 2;
            }
        }
        return bonus;
    }
}

using System.Collections.Generic;
using UnityEngine;
using UnityEngine.UI;
using TMPro;

public class UIManager : MonoBehaviour
{
    [Header("Top Stats")]
    public TextMeshProUGUI scoreText;
    public TextMeshProUGUI turnText;
    public TextMeshProUGUI energyText;
    public TextMeshProUGUI timerText;

    [Header("Field")]
    public Transform fieldRoot;
    public List<Button> zoneButtons = new List<Button>();

    [Header("Hand")]
    public Transform handRoot;
    public GameObject handCardPrefab;
    public ScrollRect handScroll;

    [Header("Card Zoom")]
    public GameObject zoomPanel;
    public TextMeshProUGUI zoomName;
    public TextMeshProUGUI zoomStats;
    public TextMeshProUGUI zoomDesc;
    public Image zoomBorder;
    public Button closeZoomButton;

    private GameManager gm;
    private Card selectedCard;
    private int selectedZone = -1;

    void Start()
    {
        gm = GameManager.Instance;
        if (gm != null)
        {
            gm.OnStateChanged += Refresh;
            gm.OnLog += _ => Refresh(gm.localPlayer);
            gm.OnCardPlaced += OnCardPlaced;
            gm.OnGoalScored += _ => Refresh(gm.localPlayer);
        }

        if (closeZoomButton != null) closeZoomButton.onClick.AddListener(HideZoom);
        HideZoom();
    }

    void OnDestroy()
    {
        if (gm != null)
        {
            gm.OnStateChanged -= Refresh;
            gm.OnGoalScored -= _ => Refresh(gm.localPlayer);
        }
    }

    public void Refresh(PlayerState p)
    {
        if (gm == null || p == null) return;

        scoreText.text = $"{gm.localPlayer.goalsScored} - {gm.remotePlayer.goalsScored}";
        turnText.text = $"{p.displayName} • {p.phase}";
        energyText.text = $"Energy {p.energy}/10";
        timerText.text = $"{p.turnTimerSeconds}s";

        RebuildHand();
        UpdateField();
    }

    void RebuildHand()
    {
        if (gm == null || gm.localPlayer == null) return;

        for (int i = handRoot.childCount - 1; i >= 0; i--) Destroy(handRoot.GetChild(i).gameObject);

        foreach (Card c in gm.localPlayer.hand)
        {
            GameObject go = Instantiate(handCardPrefab, handRoot);
            CardView view = go.GetComponent<CardView>();
            if (view != null) view.Bind(c, OnTapCard, OnDragCard);
        }
    }

    void UpdateField()
    {
        for (int i = 0; i < zoneButtons.Count && i < gm.localPlayer.field.Count; i++)
        {
            int idx = i;
            zoneButtons[i].onClick.RemoveAllListeners();
            zoneButtons[i].onClick.AddListener(() => OnZoneTapped(idx));
        }
    }

    void OnTapCard(Card c)
    {
        selectedCard = c;
        ShowZoom(c);
    }

    void OnDragCard(Card c)
    {
        selectedCard = c;
    }

    void OnZoneTapped(int idx)
    {
        selectedZone = idx;
        if (selectedCard != null)
        {
            if (gm.PlaceCard(gm.localPlayer, selectedCard, idx))
            {
                selectedCard = null;
                selectedZone = -1;
                HideZoom();
            }
        }
    }

    public void OnAttackButton()
    {
        if (selectedZone < 0) return;
        gm.Attack(gm.localPlayer, selectedZone, gm.remotePlayer, selectedZone);
    }

    void ShowZoom(Card c)
    {
        if (zoomPanel == null || c == null) return;
        zoomPanel.SetActive(true);
        zoomName.text = c.name;
        zoomStats.text = c.DisplayStatLine();
        zoomDesc.text = $"{c.description}\n\nClub: {c.club}\nNation: {c.nation}\nRarity: {c.rarity} • {c.stars}★";
        if (zoomBorder != null) zoomBorder.color = c.RarityColor();
    }

    void HideZoom()
    {
        if (zoomPanel != null) zoomPanel.SetActive(false);
    }

    void OnCardPlaced(PlayerState p, Card c)
    {
        if (p == gm.localPlayer) Refresh(p);
    }
}

public class CardView : MonoBehaviour
{
    public TextMeshProUGUI title;
    public TextMeshProUGUI statLine;
    public Image rarityFrame;
    private Card boundCard;
    private System.Action<Card> tapCallback;
    private System.Action<Card> dragCallback;

    public void Bind(Card c, System.Action<Card> onTap, System.Action<Card> onDrag)
    {
        boundCard = c;
        tapCallback = onTap;
        dragCallback = onDrag;
        if (title != null) title.text = c.name;
        if (statLine != null) statLine.text = c.type == CardType.Player ? c.DisplayStatLine() : c.description;
        if (rarityFrame != null) rarityFrame.color = c.RarityColor();
    }

    public void OnTap() { tapCallback?.Invoke(boundCard); }
    public void OnDragStart() { dragCallback?.Invoke(boundCard); }
}
