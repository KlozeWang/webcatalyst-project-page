2. ScreenCaptureService.kt - Screen Capture & Encoding

class ScreenCaptureService : Service() {

    private var mediaCodec: MediaCodec? = null
    private var virtualDisplay: VirtualDisplay? = null
    private var outputSurface: Surface? = null
    private var isRunning = false
    
    // Hardware encoding configuration
    private companion object {
        const val BIT_RATE = 8000000       // 8 Mbps
        const val FRAME_RATE = 60         // 60 FPS
        const val I_FRAME_INTERVAL = 10   // I-frame every 10s
        const val COLOR_FORMAT = 
            MediaCodecInfo.CodecCapabilities.COLOR_FormatSurface
        const val MIME_TYPE = "video/avc"   // H.264
    }

    fun startCapture(
        mediaProjection: MediaProjection,
        width: Int = 1920,
        height: Int = 1080,
        dpi: Int = 420
    ) {
        initializeEncoder(width, height)
        
        virtualDisplay = mediaProjection.createVirtualDisplay(
            "ScreenMirror",
            width, height, dpi,
            DisplayManager.VIRTUAL_DISPLAY_FLAG_AUTO_MIRROR,
            outputSurface,
            null, null
        )
        
        isRunning = true
        startEncodingLoop()
    }

    private fun initializeEncoder(width: Int, height: Int) {
        mediaCodec = MediaCodec.createEncoderByType(MIME_TYPE)
        val format = MediaFormat.createVideoFormat(MIME_TYPE, width, height).apply {
            setInteger(MediaFormat.KEY_COLOR_FORMAT, COLOR_FORMAT)
            setInteger(MediaFormat.KEY_BIT_RATE, BIT_RATE)
            setInteger(MediaFormat.KEY_FRAME_RATE, FRAME_RATE)
            setInteger(MediaFormat.KEY_I_FRAME_INTERVAL, I_FRAME_INTERVAL)
            setInteger(MediaFormat.KEY_BITRATE_MODE, 
                MediaCodecInfo.EncoderCapabilities.BITRATE_MODE_CBR)
        }
        
        mediaCodec?.configure(format, null, null, 
            MediaCodec.CONFIGURE_FLAG_ENCODE)
        
        outputSurface = mediaCodec?.createInputSurface()
        mediaCodec?.start()
    }

    private fun startEncodingLoop() {
        thread(name = "EncodingThread") {
            val codec = mediaCodec ?: return@thread
            val timeoutUs = 10000L
            
            while (isRunning) {
                val inputBufIndex = codec.dequeueInputBuffer(timeoutUs)
                if (inputBufIndex >= 0) {
                    // Buffer is ready, virtualDisplay fills it
                    codec.queueInputBuffer(
                        inputBufIndex, 0, 0, 
                        System.currentTimeMillis() * 1000, 0
                    )
                }
                
                val outputBufIndex = codec.dequeueOutputBuffer(
                    bufferInfo, timeoutUs
                )
                
                when (outputBufIndex) {
                    MediaCodec.INFO_OUTPUT_FORMAT_CHANGED -> {
                        handleFormatChanged(codec.outputFormat)
                    }
                    MediaCodec.INFO_TRY_AGAIN_LATER -> {
                        // No output available
                    }
                    else -> {
                        sendEncodedFrame(codec, outputBufIndex)
                        codec.releaseOutputBuffer(outputBufIndex, false)
                    }
                }
            }
        }
    }

    private fun sendEncodedFrame(
        codec: MediaCodec, 
        index: Int
    ) {
        val buffer = codec.getOutputBuffer(index) ?: return
        val info = bufferInfo
        
        // Send via TCP socket
        clientSocket?.outputStream.write(byteArrayOf(
            0xAA.toByte(), 0x55.toByte() // Frame header
        ))
        clientSocket?.outputStream.write(
            info.size.toByteArray()
        )
        clientSocket?.outputStream.write(buffer)
        clientSocket?.outputStream.flush()
    }

3. InputRelayService.kt - Reverse Control (Accessibility)

@AccessibilityService
class InputRelayService : AccessibilityService() {

    private var clientSocket: Socket? = null
    
    // Input event types mapping
    enum class InputType {
        TOUCH, SWIPE, KEY_EVENT, SCROLL, NAVIGATION
    }

    override fun onAccessibilityEvent(event: AccessibilityEvent?) {
        // Forward accessibility events to TV if needed
    }

    override fun onInterrupt() {
        // Handle interruption
    }

    // Handle incoming input commands from TV
    fun handleInputCommand(command: ByteArray) {
        val reader = DataInputStream(
            ByteArrayInputStream(command)
        )
        
        try {
            val type = reader.readInt()
            val x = reader.readInt()
            val y = reader.readInt()
            val action = reader.readInt()
            
            when (type) {
                InputType.TOUCH.ordinal -> {
                    performTouchInput(x, y, action)
                }
                InputType.KEY_EVENT.ordinal -> {
                    performKeyEvent(x, action)
                }
                InputType.SWIPE.ordinal -> {
                    performSwipe(x, y, reader.readInt())
                }
            }
        } catch (e: IOException) {
            Log.e("InputRelay", "Error processing input", e)
        }
    }

    private fun performTouchInput(
        x: Int, y: Int, action: Int
    ) {
        val movementEvent = MotionEvent.obtain(
            System.uptimeMillis(),
            System.uptimeMillis(),
            action, x.toFloat(), y.toFloat(), 0
        )
        
        // Dispatch to current focused window
        val window = getCurrentWindow()
        window?.decorView?.dispatchTouchEvent(movementEvent)
        
        movementEvent.recycle()
    }

    private fun performKeyEvent(
        keyCode: Int, action: Int
    ) {
        val keyEvent = KeyEvent.obtain(
            System.uptimeMillis(),
            System.uptimeMillis(),
            action, keyCode, 0,
            KeyEvent.META_OFF
        )
        
        val window = getCurrentWindow()
        window?.decorView?.dispatchKeyEvent(keyEvent)
        
        keyEvent.recycle()
    }

    private fun performSwipe(
        startX: Int, startY: Int, endX: Int
    ) {
        val endY = startY
        val duration = 300L
        
        val downEvent = MotionEvent.obtain(
            System.uptimeMillis(), 
            System.uptimeMillis(),
            MotionEvent.ACTION_DOWN,
            startX.toFloat(), startY.toFloat(), 0
        )
        
        val upEvent = MotionEvent.obtain(
            System.uptimeMillis() + duration,
            System.uptimeMillis() + duration,
            MotionEvent.ACTION_UP,
            endX.toFloat(), endY.toFloat(), 0
        )
        
        getCurrentWindow()?.decorView?.dispatchTouchEvent(downEvent)
        getCurrentWindow()?.decorView?.dispatchTouchEvent(upEvent)
        
        downEvent.recycle()
        upEvent.recycle()
    }

4. ScreenOffManager.kt - Screen Off While Streaming

class ScreenOffManager private(val context: Context) {

    private var powerManager: PowerManager? = null
    private var wakeLock: PowerManager.WakeLock? = null
    private var windowManager: WindowManager? = null
    private var isScreenOff = false

    fun initialize() {
        powerManager = context.getSystemService(
            Context.POWER_SERVICE
        ) as PowerManager
        
        windowManager = context.getSystemService(
            Context.WINDOW_SERVICE
        ) as WindowManager
    }

    fun turnScreenOff() {
        // Keep CPU running but turn off display
        wakeLock = powerManager?.newWakeLock(
            PowerManager.PARTIAL_WAKE_LOCK,
            "ScreenMirror::WakeLock"
        ).apply {
            acquire(30 * 60 * 1000L) // 30 min timeout
        }

        // Hide status bar and navigation bar
        val activity = context as Activity
        activity.window.addFlags(
            WindowManager.LayoutParams.FLAG_KEEP_SCREEN_ON or
            WindowManager.LayoutParams.FLAG_FULLSCREEN or
            WindowManager.LayoutParams.FLAG_LAYOUT_NO_LIMITS
        )

        // Create a transparent overlay to prevent screen from turning on
        val params = WindowManager.LayoutParams().apply {
            width = WindowManager.LayoutParams.MATCH_PARENT
            height = WindowManager.LayoutParams.MATCH_PARENT
            type = WindowManager.LayoutParams.TYPE_APPLICATION_OVERLAY
            flags = WindowManager.LayoutParams.FLAG_NOT_TOUCHABLE or
                    WindowManager.LayoutParams.FLAG_LAYOUT_IN_SCREEN
            format = PixelFormat.TRANSLUCENT
        }

        val overlayView = View(context).apply {
            alpha = 0.01f // Nearly invisible
            background = Color.TRANSPARENT.getDrawable()
        }

        windowManager?.addView(overlayView, params)
        isScreenOff = true
    }

    fun turnScreenOn() {
        wakeLock?.release()
        isScreenOff = false
        
        val activity = context as Activity
        activity.window.clearFlags(
            WindowManager.LayoutParams.FLAG_KEEP_SCREEN_ON or
            WindowManager.LayoutParams.FLAG_FULLSCREEN
        )
    }

    fun isScreenOff(): Boolean = isScreenOff
}

class TVMainActivity : LeanbackActivity() {

    private lateinit var tcpSocket: Socket
    private lateinit var udpSocket: DatagramSocket
    private var isConnected = false
    
    // Hardware decoder
    private lateinit var mediaCodec: MediaCodec
    private lateinit var surfaceView: SurfaceView
    private lateinit var surface: Surface
    
    // Input relay
    private lateinit var inputSender: InputSender
    
    // CPU monitor
    private val cpuMonitorRunnable = object : Runnable {
        override fun run() {
            val cpuUsage = getTVCpuUsage()
            updateCpuDisplay(cpuUsage, "tv")
            handler.postDelayed(this, 1000)
        }
    }

    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        setContentView(R.layout.activity_tv_main)
        
        surfaceView = findViewById(R.id.surfaceView)
        surface = surfaceView.surfaceHolder.addCallback(
            object : SurfaceHolder.Callback {
                override fun surfaceCreated(holder: SurfaceHolder) {
                    surface = holder.surface
                    initializeDecoder()
                }
            }
        )
        
        startCpuMonitor()
        setupInputListeners()
    }

    fun connectToMobile(view: View) {
        thread(name = "ConnectThread") {
            try {
                // Connect to mobile's hotspot IP
                tcpSocket = Socket("192.168.43.1", 8080)
                udpSocket = DatagramSocket()
                
                updateStatus("Connected to Mobile")
                isConnected = true
                
                startVideoDecoder()
                startInputRelay()
                
            } catch (e: IOException) {
                updateStatus("Connection failed: ${e.message}")
            }
        }
    }

    private fun initializeDecoder() {
        val format = MediaFormat.createVideoFormat(
            "video/avc", 1920, 1080
        ).apply {
            setInteger(MediaFormat.KEY_COLOR_FORMAT,
                MediaCodecInfo.CodecCapabilities.COLOR_FormatSurface)
        }
        
        mediaCodec = MediaCodec.createDecoderByType("video/avc")
        mediaCodec.configure(format, surface, null, 0)
        mediaCodec.start()
    }

    private fun startVideoDecoder() {
        thread(name = "DecoderThread") {
            val inputStream = tcpSocket.inputStream
            val buffer = ByteArray(8192)
            val codec = mediaCodec
            
            while (isConnected) {
                val bytesRead = inputStream.read(buffer)
                if (bytesRead > 0) {
                    decodeFrame(buffer, bytesRead)
                }
            }
        }
    }

    private fun decodeFrame(
        data: ByteArray, length: Int
    ) {
        val codec = mediaCodec
        val inputBufIndex = codec.dequeueInputBuffer(10000L)
        
        if (inputBufIndex >= 0) {
            val inputBuffer = codec.getInputBuffer(inputBufIndex)
            inputBuffer?.put(data, 0, length)
            
            codec.queueInputBuffer(
                inputBufIndex, 0, length,
                System.currentTimeMillis() * 1000, 0
            )
        }
        
        val outputBufIndex = codec.dequeueOutputBuffer(
            bufferInfo, 10000L
        )
        
        if (outputBufIndex >= 0) {
            codec.releaseOutputBuffer(outputBufIndex, true)
        }
    }

    private fun setupInputListeners() {
        // Handle TV remote input
        findViewById<View>(R.id.surfaceView)
            .setOnKeyListener { v, keyCode, event ->
                if (event.action == MotionEvent.ACTION_DOWN) {
                    sendInputToMobile(
                        InputType.KEY_EVENT, 
                        keyCode, 0, 
                        event.action
                    )
                    true
                } else {
                    false
                }
            }
        
        // Handle touch/mouse input
        findViewById<View>(R.id.surfaceView)
            .setOnTouchListener { v, event ->
                sendInputToMobile(
                    InputType.TOUCH,
                    event.x.toInt(),
                    event.y.toInt(),
                    event.action
                )
                true
            }
    }

    private fun sendInputToMobile(
        type: InputType,
        x: Int,
        y: Int,
        action: Int
    ) {
        val outputStream = tcpSocket.outputStream
        val data = ByteArray(16).apply {
            putInt(0, type.ordinal)
            putInt(4, x)
            putInt(8, y)
            putInt(12, action)
        }
        outputStream.write(data)
        outputStream.flush()
    }
}

class InputSender private(val socket: Socket) {

    enum class InputType {
        TOUCH, SWIPE, KEY_EVENT, SCROLL, NAVIGATION
    }

    fun sendTouchEvent(
        x: Float, y: Float, action: Int
    ) {
        val data = ByteArray(20).apply {
            putInt(0, InputType.TOUCH.ordinal)
            putInt(4, x.toInt())
            putInt(8, y.toInt())
            putInt(12, action)
            putFloat(16, pressure)
        }
        send(data)
    }

    fun sendKeyEvent(
        keyCode: Int, action: Int
    ) {
        val data = ByteArray(16).apply {
            putInt(0, InputType.KEY_EVENT.ordinal)
            putInt(4, keyCode)
            putInt(8, action)
        }
        send(data)
    }

    fun sendSwipeEvent(
        startX: Float, startY: Float,
        endX: Float, endY: Float
    ) {
        val data = ByteArray(24).apply {
            putInt(0, InputType.SWIPE.ordinal)
            putInt(4, startX.toInt())
            putInt(8, startY.toInt())
            putInt(12, endX.toInt())
            putInt(16, endY.toInt())
        }
        send(data)
    }

    private fun send(data: ByteArray) {
        try {
            socket.outputStream.write(data)
            socket.outputStream.flush()
        } catch (e: IOException) {
            Log.e("InputSender", "Failed to send input", e)
        }
    }
}

Protocol Specification

/*
 * ============================================
 * SCREEN MIRRORING PROTOCOL SPECIFICATION
 * ============================================
 * 
 * Transport: TCP (video) + UDP (input/controls)
 * Video Codec: H.264 (AVC) / H.265 (HEVC)
 * Frame Rate: 30-60 FPS
 * Resolution: Up to 1920x1080
 * Bitrate: 4-16 Mbps (adaptive)
 * Latency Target: < 50ms
 */

// === TCP Protocol (Video Stream) ===

// Frame Header Structure (16 bytes)
struct FrameHeader {
    uint32  magic;        // 0xAA55AA55
    uint32  frameNumber;  // Sequential frame counter
    uint32  dataSize;     // Size of payload in bytes
    uint32  timestamp;    // Unix timestamp in microseconds
}

// Video Frame Payload
struct VideoFrame {
    FrameHeader header;
    byte[]      payload;  // Encoded H.264 NAL units
}

// === UDP Protocol (Input Commands) ===

// Input Command Structure (32 bytes)
struct InputCommand {
    uint8   type;         // 0=TOUCH, 1=KEY, 2=SWIPE, 3=SCROLL
    uint8   action;       // DOWN, UP, MOVE
    int16   x;            // X coordinate
    int16   y;            // Y coordinate
    uint32  keyCode;      // For KEY events
    uint32  timestamp;    // Timestamp in microseconds
    uint16  padding;      // Alignment padding
}

// === Control Messages (TCP) ===

enum ControlMessage : uint8 {
    CONNECT_REQUEST = 0x01,
    CONNECT_RESPONSE = 0x02,
    DISCONNECT = 0x03,
    SCREEN_OFF = 0x04,
    SCREEN_ON = 0x05,
    HEARTBEAT = 0x06,
    QUALITY_CHANGE = 0x07,
    DEVICE_INFO = 0x08,
}

// === Connection Handshake ===

// 1. Mobile starts TCP server on port 8080
// 2. TV connects to Mobile's IP:8080
// 3. TV sends CONNECT_REQUEST with device info
// 4. Mobile responds with CONNECT_RESPONSE + capabilities
// 5. Video stream begins

// === Heartbeat (every 5 seconds) ===
struct Heartbeat {
    uint8   type = HEARTBEAT;
    uint32  mobileCpu;    // Mobile CPU usage %
    uint32  tvCpu;        // TV CPU usage %
    uint32  latency;      // Round-trip latency in ms
    uint32  bitrate;      // Current bitrate in kbps
}

// === Quality Adaptation ===
// If latency > 100ms for 3+ seconds:
//   → Reduce bitrate by 25%
//   → Reduce frame rate to 30 FPS
// If latency < 30ms for 5+ seconds:
//   → Increase bitrate by 25%
//   → Increase frame rate to 60 FPS

⚙️ Step-by-Step Setup Guide

Prerequisites

  • Android Mobile Device

    Android 8.0 (API 26) or higher. Must support MediaProjection API. Recommended: Android 12+ for best H.265 support.

  • Android TV Device

    Android TV 9.0 (API 28) or higher. Must support hardware H.264/H.265 decoding. Android TV Emulator or physical TV box.

  • Development Environment

    Android Studio Hedgehog or later. Kotlin 1.9+. Minimum SDK 26. Target SDK 34.

Installation Steps

  • Step 1: Clone & Build Mobile App

    Open the mobile app project in Android Studio. Sync Gradle files. Ensure all dependencies are installed. Build and install the APK on your mobile device.

  • Step 2: Clone & Build TV App

    Open the TV app project in Android Studio. Sync Gradle files. Build and install the APK on your Android TV device or emulator.

  • Step 3: Enable Accessibility Service

    On the mobile device, go to Settings → Accessibility → Installed Services → Enable "Screen Mirror Input Relay". Grant the required permissions.

  • Step 4: Configure Network

    On the mobile device, enable Personal Hotspot. Note the hotspot IP address (typically 192.168.43.1). Connect the Android TV to this hotspot via Wi-Fi.

  • Step 5: Start Mirroring

    Open the mobile app and tap "Start Server". On the TV app, tap "Connect to Mobile". The mobile screen should now appear on the TV with reverse control enabled.

  • Step 6: Enable Screen Off Mode (Optional)

    Toggle "Screen Off Mode" on the mobile app to save battery. The screen will turn off but streaming and processing continue in the background.

Troubleshooting

❌

Connection Failed

Ensure both devices are on the same network. Check firewall settings. Verify the mobile hotspot is active and TV is connected to it.

🐌

High Latency

Reduce video bitrate in settings. Ensure strong Wi-Fi signal. Close other network-intensive apps on both devices.

📺

Black Screen on TV

Check if MediaProjection permission is granted. Ensure the mobile screen is not locked. Try restarting both apps.

🎮

Input Not Working

Verify Accessibility Service is enabled. Check UDP port 8081 is not blocked. Ensure TV remote/mouse input is properly mapped.
