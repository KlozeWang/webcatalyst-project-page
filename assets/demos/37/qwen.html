package com.screenmirror.mobile import android.os.Build import java.io.BufferedReader import java.io.FileReader object CpuMonitor { fun getCpuUsage(): Float { return try { val values = readCpuStats() val total1 = values[0] + values[1] + values[2] + values[3] + values[4] + values[5] + values[6] val idle1 = values[4] Thread.sleep(500) val values2 = readCpuStats() val total2 = values2[0] + values2[1] + values2[2] + values2[3] + values2[4] + values2[5] + values2[6] val idle2 = values2[4] val totalDiff = total2 - total1 val idleDiff = idle2 - idle1 if (totalDiff == 0) 0f else ((totalDiff - idleDiff) / totalDiff) * 100f } catch (e: Exception) { 0f } } private fun readCpuStats(): FloatArray { val reader = BufferedReader(FileReader("/proc/stat")) val line = reader.readLine() reader.close() val parts = line?.split("\\s+") ?: emptyList() return FloatArray(7) { index -> parts.getOrNull(index + 1)?.toFloatOrNull() ?: 0f } } }
